#!/bin/bash
# Deploy script para Controle de Débitos na VPS Êxito
# Host: root@179.199.149.12, Repo: /root/PROJETOS/exito/controle-de-debitos, PM2: exito-debitos
# Backup dos 3 JSON: empresas.json, cadastro-consultas.json, parcelamentos.json
# Nota: NÃO rodar git clean -fd (apaga PDFs e inbox)
# Python: o do sistema (Ubuntu, sem venv). Libs de PDF via pip --break-system-packages;
# Tesseract via apt (tesseract-ocr tesseract-ocr-por), instalado uma vez.
# Rodar colando no terminal da VPS. Via pipe do PowerShell os acentos do caminho viram "?".

set -euo pipefail

cd /root/PROJETOS/exito/controle-de-debitos

APP_DIR="2. RELAÇÃO DE DEBITOS"
DATA_DIR="$APP_DIR/dashboard/data"
BACKUP_DIR="/tmp/debitos-data-backup"
mkdir -p "$BACKUP_DIR"

echo "=== BACKUP ==="
# 1) guarda os dados reais do servidor (não podem ir embora no pull)
cp -a "$DATA_DIR/empresas.json" "$BACKUP_DIR/empresas.json"
cp -a "$DATA_DIR/cadastro-consultas.json" "$BACKUP_DIR/cadastro-consultas.json"
cp -a "$DATA_DIR/parcelamentos.json" "$BACKUP_DIR/parcelamentos.json"
echo "JSON backed up to $BACKUP_DIR"

echo "=== NO-SKIP-WORKTREE ==="
# 2) com skip-worktree ligado o git recusa o merge quando o JSON muda no commit novo
git update-index --no-skip-worktree \
  "$DATA_DIR/empresas.json" \
  "$DATA_DIR/cadastro-consultas.json" \
  "$DATA_DIR/parcelamentos.json"

echo "=== GIT RESTORE ==="
# 3) libera o merge: descarta o diff local dos JSON (estão no backup) e dos .pyc (cache regenerável)
git restore -- \
  "$DATA_DIR/empresas.json" \
  "$DATA_DIR/cadastro-consultas.json" \
  "$DATA_DIR/parcelamentos.json" \
  "$APP_DIR/scripts/__pycache__"
git restore -- "$APP_DIR/dashboard/.next" 2>/dev/null || true

rm -rf "$APP_DIR/dashboard/.next/server/app/_not-found"

echo "=== GIT PULL ==="
# 4) atualiza só o código/interface
git fetch origin
git pull --ff-only origin main

echo "=== RESTORE PRODUCTION JSON ==="
# 5) devolve os dados do sistema (painel + consultas + parcelamentos)
cp -a "$BACKUP_DIR/empresas.json" "$DATA_DIR/empresas.json"
cp -a "$BACKUP_DIR/cadastro-consultas.json" "$DATA_DIR/cadastro-consultas.json"
cp -a "$BACKUP_DIR/parcelamentos.json" "$DATA_DIR/parcelamentos.json"
echo "JSON restored and verified"

echo "=== SKIP-WORKTREE ==="
# 6) pull futuro não sobrescreve esses arquivos
git update-index --skip-worktree \
  "$DATA_DIR/empresas.json" \
  "$DATA_DIR/cadastro-consultas.json" \
  "$DATA_DIR/parcelamentos.json"

echo "=== PYTHON DEPS ==="
# 7) sem pymupdf/pypdf o servidor não lê o texto do PDF (card de diagnóstico fiscal some)
python3 -m pip install --break-system-packages --root-user-action=ignore -q \
  -r "$APP_DIR/scripts/requirements-debitos.txt"
python3 -c "import pypdf, pdfplumber, fitz, pytesseract, PIL; print('libs PDF OK')"

echo "=== BUILD ==="
cd "$APP_DIR/dashboard"
npm run build

echo "=== PM2 RESTART ==="
pm2 restart exito-debitos
sleep 3
pm2 show exito-debitos | head -25

echo "=== HTTP CHECK ==="
curl -s -o /dev/null -w "GET / : %{http_code}\n" http://127.0.0.1:6999/
curl -s -o /dev/null -w "GET /upload : %{http_code}\n" http://127.0.0.1:6999/upload
curl -s -o /dev/null -w "GET /consultas : %{http_code}\n" http://127.0.0.1:6999/consultas
curl -s -o /dev/null -w "GET /parcelamentos : %{http_code}\n" http://127.0.0.1:6999/parcelamentos

echo "=== DEPLOY OK ==="
cd /root/PROJETOS/exito/controle-de-debitos
git rev-parse --short HEAD
git log -1 --format='%h %s'
